      function moverArrastreTactil(evento) {
        if (
          !estadoTactil.activo ||

          evento.clientX - estadoTactil.inicioX

        var diferenciaY = Math.abs(
          evento.clientY - estadoTactil.inicioY
        );

        if (
          !estadoTactil.movido &&
          diferenciaX < 8 &&
          diferenciaY < 8
        ) {

        estadoTactil.movido = true;

        if (!estadoTactil.fantasma) {
          estadoTactil.fantasma = crearFantasma(
            estadoTactil.tarjeta,
            evento.clientX,
            evento.clientY

          evento.clientX + "px";

        estadoTactil.fantasma.style.top =
          evento.clientY + "px";

          buscarDestinoEnPunto(
            evento.clientX,
            evento.clientY
          )
        );
      }

      function finalizarArrastreTactil(evento) {

          estadoTactil.movido &&

          estadoTactil.fantasma &&
          estadoTactil.fantasma.parentNode

            estadoTactil.fantasma
          );
        }

        estadoTactil.activo = false;
        estadoTactil.tarjeta = null;
        estadoTactil.fantasma = null;
        estadoTactil.pointerId = null;
        estadoTactil.movido = false;
